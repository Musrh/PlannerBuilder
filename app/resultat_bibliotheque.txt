                  return
                }
                const reader = new FileReader()
                reader.onload = () => {
                  setCustomPatternImage(String(reader.result || ''))
                  setMessage('Image de motif importée.')
                }
                reader.onerror = () => setMessage('Impossible de lire cette image de motif.')
                reader.readAsDataURL(f)
              }}
            />
          </>

                  <strong title={pattern.name}>{pattern.name}</strong>

            onClick={downloadMockup}

            Télécharger le mockup {downloadFormat.toUpperCase()}
          </button>
