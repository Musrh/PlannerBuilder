            <label className="field-label" htmlFor="pages">Nombre de pages</label>
            <div className="number-control">
              <button type="button" onClick={() => setPages((value) => Math.max(1, value - 1))} aria-label="Diminuer le nombre de pages">−</button>
              <input id="pages" type="number" min="1" max="365" value={pages} onChange={(event) => setPages(Math.max(1, Math.min(365, Number(event.target.value) || 1)))} />
              <button type="button" onClick={() => setPages((value) => Math.min(365, value + 1))} aria-label="Augmenter le nombre de pages">+</button>
            </div>
            <small className="helper">De 1 à 365 pages dans cette version de démonstration.</small>
          </div>

          <div className="setting-group">
            <span className="field-label">Couleur principale</span>
            <div className="color-control">
              <label className="color-swatch" style={{ background: primary }} htmlFor="primary-color" />
              <input id="primary-color" type="color" value={primary} onChange={(event) => setPrimary(event.target.value)} />
              <code>{primary.toUpperCase()}</code>
            </div>
          </div>

          <div className="setting-group">
            <span className="field-label">Fond des pages</span>
            <div className="color-control">
              <label className="color-swatch paper-swatch" style={{ background: background }} htmlFor="background-color" />
              <input id="background-color" type="color" value={background} onChange={(event) => setBackground(event.target.value)} />
              <code>{background.toUpperCase()}</code>
            </div>
          </div>

          <div className="setting-group">
            <label className="field-label" htmlFor="font">Typographie</label>
            <select id="font" value={font} onChange={(event) => setFont(event.target.value)}>
              <option value="moderne">Moderne</option>
              <option value="classique">Classique</option>
              <option value="elegante">Élégante</option>
            </select>
          </div>

          <div className="setting-group">
            <span className="field-label">Densité de la page</span>
            <div className="segmented">
              {['compacte', 'confortable'].map((value) => (
                <button key={value} type="button" className={density === value ? 'active' : ''} onClick={() => setDensity(value)}>
                  {value === 'compacte' ? 'Compacte' : 'Aérée'}
                </button>
              ))}
            </div>
          </div>

          <button className="reset-button" type="button" onClick={() => {
            setPrimary('#718B70')
            setBackground('#F7F5EF')
            setFont('moderne')
            setDensity('confortable')
            setNotice('')
          }}>
            ↺ Réinitialiser le style
          </button>

          <div className="export-area">
            <button className="export-button" type="button" onClick={exportPDF}>
              <span>↓</span> Exporter en PDF
            </button>
