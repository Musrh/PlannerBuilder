function App() {
  const { user, plan, isAdmin } = usePlannerBuilderAuth()
  const isUnlimited = plan === 'pro' || isAdmin

  // Compteur d'exports gratuits propre à chaque compte Firebase
  const exportStorageKey = user?.uid
    ? `plannerbuilder_free_exports_${user.uid}`
    : null

  const [showMockups, setShowMockups] = useState(false)
  const [showAdminPanel, setShowAdminPanel] = useState(false)
  const [selectedId, setSelectedId] = useState('food')
  const [visualStyle, setVisualStyle] = useState('minimaliste')
  const [layoutModel, setLayoutModel] = useState('classique')
  const [coverId, setCoverId] = useState('')
  const [coverImage, setCoverImage] = useState('')
  const coverEditorRef = useRef(null)
  const [coverMode, setCoverMode] = useState('preset')
  const [pages, setPages] = useState(12)
  const [primary, setPrimary] = useState('#718B70')
  const [background, setBackground] = useState('#F7F5EF')
  const [titleBackground, setTitleBackground] = useState('#718B70')
  const [titleTextColor, setTitleTextColor] = useState('#FFFFFF')
  const [font, setFont] = useState('moderne')
  const [density, setDensity] = useState('confortable')
  const [category, setCategory] = useState('Tous')
  const [notice, setNotice] = useState('')

  function readFreeExports(storageKey) {
    if (!storageKey) return 0

    try {
      const saved = Number(localStorage.getItem(storageKey) || 0)

      return Number.isFinite(saved) && saved >= 0
        ? Math.floor(saved)
        : 0
    } catch {
      return 0
    }
  }

  const [freeExports, setFreeExports] = useState(() =>
    readFreeExports(exportStorageKey)
  )

  // Recharge le compteur si un autre compte se connecte sur le même navigateur
  useEffect(() => {
    setFreeExports(readFreeExports(exportStorageKey))
  }, [exportStorageKey])

  const selected = templates.find((item) => item.id === selectedId)
  const content = pageContent[selectedId]
  const categories = ['Tous', ...new Set(templates.map((item) => item.category))]
  const filtered = category === 'Tous'
    ? templates
    : templates.filter((item) => item.category === category)

  function chooseTemplate(template) {
    setSelectedId(template.id)
    setPages(template.pages)
    setNotice('')
  }

  async function exportPDF() {
    if (coverMode === 'preset' && !coverId && !coverImage) {
      setNotice('Choisissez une couverture ou importez votre propre image avant l’export.')
      return
    }

    if (!isUnlimited && !exportStorageKey) {
      setNotice('Connectez-vous pour exporter votre PDF.')
      return
    }

    if (!isUnlimited && freeExports >= 3) {
      setNotice(
        'Vous avez utilisé vos 3 exports gratuits. Passez à PlannerBuilder Pro pour exporter sans limite.'
      )
      return
    }

    try {
      const doc = new jsPDF({
        orientation: 'portrait',
        unit: 'mm',
        format: 'a4',
      })

      const pageWidth = 210
      const pageHeight = 297
      const margin = 15
      const usableWidth = pageWidth - margin * 2

      const toRgb = (hex) => {
        const value = hex.replace('#', '')
        return [
          parseInt(value.substring(0, 2), 16),
          parseInt(value.substring(2, 4), 16),
          parseInt(value.substring(4, 6), 16),
        ]
      }

      const primaryRgb = toRgb(primary)
      const backgroundRgb = toRgb(background)
      const titleBackgroundRgb = toRgb(titleBackground)
      const titleTextRgb = toRgb(titleTextColor)

      const fontMap = {
        moderne: 'helvetica',
        classique: 'times',
        elegante: 'times',
      }

      const pdfFont = fontMap[font] || 'helvetica'
      const compact = density === 'compacte'
      const sectionGap = compact ? 5 : 8
      const lineGap = compact ? 6 : 8

      // PAGE 1 : couverture indépendante du style intérieur.
      const coverThemes = {
        minimaliste: { bg: [250, 249, 246], accent: [55, 55, 55], shape: 'minimal' },
        nature: { bg: [237, 242, 231], accent: [74, 112, 76], shape: 'nature' },
        professionnelle: { bg: [238, 242, 248], accent: [39, 63, 95], shape: 'professional' },
        coloree: { bg: [255, 239, 220], accent: [190, 75, 67], shape: 'color' },
        elegante: { bg: [247, 241, 235], accent: [128, 91, 72], shape: 'elegant' },
        florale: { bg: [250, 237, 242], accent: [160, 83, 119], shape: 'floral' },
        geometrique: { bg: [237, 239, 247], accent: [70, 75, 150], shape: 'geometric' },
        sombre: { bg: [35, 38, 43], accent: [225, 192, 125], shape: 'dark' },
      }

      const cover = coverThemes[coverId]

      if (coverMode === 'custom') {
        if (!coverEditorRef.current) {
          throw new Error('Éditeur de couverture indisponible.')
        }

        const customImage = await coverEditorRef.current.exportImage()
        doc.addImage(customImage, 'PNG', 0, 0, pageWidth, pageHeight)
      } else if (coverImage) {
        const format = coverImage.startsWith('data:image/png')
          ? 'PNG'
          : coverImage.startsWith('data:image/webp')
            ? 'WEBP'
            : 'JPEG'

        doc.addImage(coverImage, format, 0, 0, pageWidth, pageHeight)
        doc.setFillColor(255, 255, 255)
        doc.roundedRect(18, 95, pageWidth - 36, 90, 4, 4, 'F')
        doc.setTextColor(30, 30, 30)
      } else {
        if (!cover) {
          throw new Error('Veuillez choisir une couverture prédéfinie.')
        }

        doc.setFillColor(...cover.bg)
        doc.rect(0, 0, pageWidth, pageHeight, 'F')
        doc.setFillColor(...cover.accent)

        if (cover.shape === 'nature') {
          doc.circle(175, 45, 30, 'F')
          doc.circle(25, 245, 35, 'F')
        } else if (cover.shape === 'professional') {
          doc.rect(0, 0, 12, pageHeight, 'F')
          doc.rect(25, 35, 160, 1.5, 'F')
        } else if (cover.shape === 'color') {
          doc.circle(170, 50, 38, 'F')
          doc.circle(35, 260, 27, 'F')
        } else if (cover.shape === 'elegant') {
          doc.setDrawColor(...cover.accent)
          doc.setLineWidth(0.8)
          doc.rect(12, 12, pageWidth - 24, pageHeight - 24, 'S')
          doc.rect(17, 17, pageWidth - 34, pageHeight - 34, 'S')
        } else if (cover.shape === 'floral') {
          doc.circle(35, 40, 18, 'F')
          doc.circle(175, 255, 25, 'F')
          doc.circle(35, 58, 8, 'F')
        } else if (cover.shape === 'geometric') {
          doc.triangle(150, 20, 205, 80, 155, 105, 'F')
          doc.rect(0, 230, pageWidth, 67, 'F')
        } else if (cover.shape === 'dark') {
          doc.setDrawColor(...cover.accent)
          doc.setLineWidth(1.2)
          doc.rect(12, 12, pageWidth - 24, pageHeight - 24, 'S')
        } else {
          doc.rect(20, 35, 42, 1, 'F')
          doc.rect(20, 40, 24, 1, 'F')
        }

        doc.setTextColor(...cover.accent)
      }

      if (coverMode === 'preset' && !coverImage) {
        doc.setFont('helvetica', 'bold')
        doc.setFontSize(11)
        doc.text('PLANNERBUILDER', pageWidth / 2, 88, { align: 'center' })

        doc.setFont(pdfFont, 'bold')
        doc.setFontSize(27)
        const coverTitle = doc.splitTextToSize(content.title, pageWidth - 42)
        doc.text(coverTitle, pageWidth / 2, 125, { align: 'center' })

        doc.setFont(pdfFont, 'normal')
        doc.setFontSize(12)
        doc.text(
          doc.splitTextToSize(content.subtitle, pageWidth - 48),
          pageWidth / 2,
          145,
          { align: 'center' }
        )

        doc.setFont('helvetica', 'normal')
        doc.setFontSize(10)
        doc.text(selected.category.toUpperCase(), pageWidth / 2, 175, { align: 'center' })
        doc.setFontSize(8)
        doc.text('COUVERTURE', pageWidth / 2, pageHeight - 18, { align: 'center' })
      }
