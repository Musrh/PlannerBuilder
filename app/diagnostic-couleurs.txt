function App() {
  const [selectedId, setSelectedId] = useState('food')
  const [pages, setPages] = useState(12)
  const [primary, setPrimary] = useState('#718B70')
  const [background, setBackground] = useState('#F7F5EF')
  const [font, setFont] = useState('moderne')
  const [density, setDensity] = useState('confortable')
  const [category, setCategory] = useState('Tous')
  const [notice, setNotice] = useState('')

  const selected = templates.find((item) => item.id === selectedId)
  const content = pageContent[selectedId]
  const categories = ['Tous', ...new Set(templates.map((item) => item.category))]
  const filtered = category === 'Tous'
    ? templates
    : templates.filter((item) => item.category === category)

  function chooseTemplate(template) {
    setSelectedId(template.id)
    setPages(template.pages)
    setNotice('')
  }

  function exportPDF() {
    try {
      const doc = new jsPDF({
        orientation: 'portrait',
        unit: 'mm',
        format: 'a4',
      })

      const pageWidth = 210
      const pageHeight = 297
      const margin = 15
      const usableWidth = pageWidth - margin * 2

      const toRgb = (hex) => {
        const value = hex.replace('#', '')
        return [
          parseInt(value.substring(0, 2), 16),
          parseInt(value.substring(2, 4), 16),
          parseInt(value.substring(4, 6), 16),
        ]
      }

      const primaryRgb = toRgb(primary)
      const backgroundRgb = toRgb(background)
      const fontMap = {
        moderne: 'helvetica',
        classique: 'times',
        elegante: 'times',
      }
      const pdfFont = fontMap[font] || 'helvetica'
      const compact = density === 'compacte'
      const sectionGap = compact ? 5 : 8
      const lineGap = compact ? 6 : 8

      for (let pageIndex = 0; pageIndex < pages; pageIndex += 1) {
        if (pageIndex > 0) doc.addPage()

        doc.setFillColor(...backgroundRgb)
        doc.rect(0, 0, pageWidth, pageHeight, 'F')

        doc.setDrawColor(...primaryRgb)
        doc.setLineWidth(0.8)
        doc.line(margin, 13, pageWidth - margin, 13)

        doc.setFont('helvetica', 'bold')
        doc.setFontSize(8)
        doc.setTextColor(...primaryRgb)
        doc.text('MON PLANNER', margin, 10)
        doc.text(String(pageIndex + 1).padStart(2, '0'), pageWidth - margin, 10, {
          align: 'right',
        })

        doc.setFont(pdfFont, 'normal')
        doc.setFontSize(10)
        doc.setTextColor(...primaryRgb)
        doc.text(selected.category.toUpperCase(), margin, 26)

        doc.setFont(pdfFont, 'bold')
        doc.setFontSize(23)
        doc.setTextColor(0, 0, 0)
        const titleLines = doc.splitTextToSize(content.title, usableWidth)
        doc.text(titleLines, margin, 37)

        let y = 37 + titleLines.length * 9
        doc.setFont(pdfFont, 'normal')
        doc.setFontSize(10)
